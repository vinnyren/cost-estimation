## 4. Phase A — 问题 1: 上传文件管理

### 4.1 backend: GET /api/uploads 增强 + DELETE

修改 `server/app/api/uploads.py`:

```python
@router.get("")
def list_uploads(project_id: str | None = None, db = Depends(get_db)):
    """v2.5 — 列表含 size + created_at。可按 project_id 过滤。"""
    q = db.query(Upload)
    if project_id:
        q = q.filter_by(project_id=project_id)
    rows = q.order_by(Upload.created_at.desc()).all()
    return [
        {
            "id": u.id,
            "project_id": u.project_id,
            "filename": u.filename,
            "size_bytes": u.size_bytes,
            "mime_type": u.mime_type,
            "created_at": u.created_at.isoformat(),
        }
        for u in rows
    ]


@router.delete("/{upload_id}", status_code=204)
def delete_upload(upload_id: str, db = Depends(get_db)):
    """v2.5 — 删除上传 + 物理文件清理。"""
    u = db.query(Upload).filter_by(id=upload_id).first()
    if not u:
        raise HTTPException(404, "upload not found")
    # 删物理文件
    from pathlib import Path
    fp = Path(u.disk_path)
    if fp.exists():
        fp.unlink()
    db.delete(u)
    db.commit()
    return None
```

> 注：实际 `Upload` model 字段以 `db/models.py` 现有为准（`size_bytes` / `disk_path` 名称如有不同按实际改）。

```typescript
  if (props.projectId) {
    // edit mode: fetch existing project 预填表单
    const proj = await projectsApi.get(props.projectId);
    Object.assign(form, {
      name: proj.name,
      city: proj.city,
      industry: proj.industry,
      phase: proj.phase,
      project_type: proj.project_type,
      mode: proj.mode,
      target_total: proj.target_cost ?? 0,
      client: proj.client ?? '',
      evaluator: proj.evaluator ?? '',
      alpha: proj.alpha_dev ?? 1.0,
      include_ops: proj.include_ops ?? false,
      factors_dev: proj.factors_dev ?? {},
      factors_ops: proj.factors_ops ?? {},
    });
  }
});

async function submit() {
  if (isEditMode.value) {
    await projectsApi.update(props.projectId!, form);
    router.push(`/projects/${props.projectId}/functions`);
  } else {
    const proj = await projectsApi.create(form);
    router.push(`/projects/${proj.id}/functions`);
  }
}
```

## 7. Phase D — 问题 5: Claude 后台调用 + 任务列表

### 7.1 services/ai_tasks.py 加 spawn

```python
import subprocess
import shutil
import os
from pathlib import Path


def spawn_claude_extract(task_id: str, project_id: str, base_url: str, token: str) -> int | None:
    """v2.5 — 后台 spawn `claude --print` 跑 /cost <project_id>.

    Returns subprocess PID; None if claude CLI 不可用。
    Backend store PID in AiTask.pid for stop/status query.
    """
    claude_bin = shutil.which("claude")
    if not claude_bin:
        return None

    # 让 Claude 跑 cost.md 命令；其 6 次 PATCH 走 BASE/TOKEN/PROJECT_ID 上报
    cmd = [
        claude_bin, "--print",
        "--allowedTools", "Bash(curl *) Bash(jq *) Read",
        f"/cost {project_id}",
    ]
    env = {
        **os.environ,
        "BASE": base_url,
        "TOKEN": token,
        "PROJECT_ID": project_id,
        "TASK_ID": task_id,  # 让 cost.md 复用我们已创建的 task
    }
    # 用 Popen detach；stdout/stderr 写到日志文件方便排查
    log_path = Path(os.environ.get("COST_DATA_DIR", "/tmp")) / f"ai-task-{task_id}.log"
    proc = subprocess.Popen(
        cmd, env=env,
        stdout=open(log_path, "w"),
        stderr=subprocess.STDOUT,
        start_new_session=True,  # 脱离 parent，不被 server 重启杀
    )
    return proc.pid


def stop_claude_subprocess(pid: int) -> bool:
    """v2.5 — kill 后台 Claude 进程。返回是否成功。"""
    try:
        os.kill(pid, signal.SIGTERM)
        return True
    except (ProcessLookupError, PermissionError):
        return False
```

```python
def start_task(task_id: str, db = Depends(get_db)):
    """v2.5 — UI 触发 spawn Claude 后台执行。"""
    t = svc.get_task(db, task_id)
    if not t:
        raise HTTPException(404, "task not found")
    if t.status in ("running", "done"):
        raise HTTPException(409, f"task already {t.status}")

    base_url = "http://127.0.0.1:8788"  # spawned claude 调本机
    token = os.environ.get("COST_AUTH_TOKEN", "")
    pid = svc.spawn_claude_extract(t.id, t.project_id, base_url, token)
    if pid is None:
        raise HTTPException(500, "claude CLI 不可用 — 请确保 claude 在 PATH 中且已登录")
    svc.update_task(db, t.id, status="running", progress_pct=1.0,
                    stage_log_append="✓ 后台进程已启动")
    t.pid = pid
    db.commit()
    return {"pid": pid}


@router.post("/{task_id}/stop")
def stop_task(task_id: str, db = Depends(get_db)):
    """v2.5 — kill 后台进程并标记 failed。"""
    t = svc.get_task(db, task_id)
    if not t:
        raise HTTPException(404, "task not found")
    if not t.pid:
        raise HTTPException(400, "task has no associated subprocess")
    ok = svc.stop_claude_subprocess(t.pid)
    svc.update_task(db, t.id, status="failed",
                    error_message="用户手动停止" if ok else "停止失败 — 进程可能已退出")
    return {"stopped": ok}
```

## 8. 数据流（问题 5 关键）

```
UI 点击 "+ 新建提取任务"
  ↓
POST /api/ai-tasks {project_id, kind:"extract"}     ← 创建 task
  ↓ task_id
POST /api/ai-tasks/{task_id}/start                  ← spawn claude 后台
  ↓
backend: subprocess.Popen([
  "claude", "--print",
  "--allowedTools", "Bash(curl *) Read",
  "/cost p-xxx"
], env={BASE, TOKEN, PROJECT_ID, TASK_ID}, start_new_session=True)
  ↓
Claude 进程后台跑 cost.md 流程:
  curl PATCH /api/ai-tasks/{TASK_ID} progress=10 ...
  curl PATCH ... progress=30 ...
  ...
  curl PATCH ... status=done
  ↓
UI 1.5s polling list → 实时显示进度
  ↓ 用户随时 [停止]
POST /api/ai-tasks/{task_id}/stop                   ← kill pid
```

## 9. 错误处理

- **claude CLI 不存在**: `/start` 返回 500 "claude CLI 不可用 — 请确保 claude 在 PATH 中且已登录"
- **claude 未登录**: spawn 成功但进程立即退出 → PATCH 上报"启动失败"
- **stop 时进程已退出**: ok=false，但 task 仍标记 failed
- **subprocess 卡死**: 没有 timeout 机制（v2.5 不做）— 用户可手动 [停止]
